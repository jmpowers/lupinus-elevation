lup.data.cut <- lup.data[lup.data$Filename %in% lupmeta$FileName,]
#lup.data.cut$Filename <- lupmeta$sample[match(lup.data.cut$Filename, lupmeta$FileName)] #don't use sample!

lup.data.cut$Name <- droplevels(lup.data.cut$Name)
longdata <- load_longdata(lup.data.cut, sample="Filename", RT="Ret.Time", name="Name", area="Area", match = "SI", maxmatch=100)
metadata <- load_metadata(lupmeta %>% rename(realsample=sample), 
                          sample="FileName", type="type")#was sample="sample"
metadata$type <- factor(metadata$type)
vol.all <- make_sampletable(longdata, metadata)
chems <- make_chemtable(longdata, metadata)

### Flower filtering
chemsf <- chems %>%
  filter_RT(2, 17) %>% 
  filter_match(0.8) %>% 
  filter_freq(0.1, group = FALSE) %>% 
  filter_contaminant(cont.list = c("Acetic acid, trichloro-, nonyl ester",
                                   "2-Pentanol, 3-chloro-4-methyl-, (R*,R*)-(.+/-.)-",
                                   "Hexyl chloroformate", "Oxirane, decyl-", #below: alkanes with many RTs
                                   "4-methylpentanoic acid", "hexadecane", "1-nonoxytetradecane",
                                   "13-methyltetradecanal", "nonylcyclohexane", "3,8-dimethyldecane",
                                   "farnesane", "2,5-dimethylnonane", "3,3,5-trimethylheptane",
                                   "3,3,5-trimethylcyclohexan-1-ol")) %>% #quant and quant never both nonzero
  filter_area(min_maximum = 1e6) %>% 
  filter_ambient_ratio(vol.all, metadata, ratio = 4) %>%
  filter_ambient_ttest(vol.all, metadata, alpha = 0.05, adjust = "fdr") %>%
  combine_filters() 
chemsf$filter_final <- with(chemsf, (filter_RT == "OK" & filter_match=="OK" & filter_freq.floral == "OK" & 
                              filter_contaminant == "OK" & filter_area == "OK" & 
                                filter_ambient_ratio == "OK") | name=="indole")

#Prune based on bouquet filters
vol <- prune_sampletable(vol.all, chemsf, metadata)
exclude_low <- c("JH_Lup_arg_BVB_24-23redo_7102025_03.qgd", "JH_Lup_arg_BVB_24-30_792025_03.qgd") #<1 ng/hr volatiles
meta <- metadata[metadata$type == "floral",][!rownames(vol) %in% exclude_low,]
vol <- vol[!rownames(vol) %in% exclude_low,] 
chm <- chemsf[match(colnames(vol), chemsf$name),]

#Convert peak areas to nanograms
lupchems2 <- lupchems[match(colnames(vol), lupchems$shortname),]
vol <- sweep(vol, 2, lupchems2$area_per_ng, FUN = '/')
vol <- vol / (40/60 + 20/60) # divide by equilibration + pumping time in hours (40+20 min)
meta$Total <- rowSums(vol)
